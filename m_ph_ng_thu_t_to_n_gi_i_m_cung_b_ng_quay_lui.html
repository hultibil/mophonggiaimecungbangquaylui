Thuật toán Quay lui (Backtracking) là gì?

Quay lui là một kỹ thuật thiết kế thuật toán dựa trên đệ quy nhằm tìm kiếm tất cả hoặc một giải pháp cho các bài toán thỏa mãn các ràng buộc cho trước. Bằng cách thử nghiệm từng bước một, khi đi đến một nhánh cụt (gặp tường hoặc đã đi qua mà không tìm được đường tới đích), thuật toán sẽ quay lui (backtrack) về bước trước đó để thử con đường hoặc hướng đi khác.

1. Tiến (Forward)

Thử di chuyển theo các hướng khả dĩ (Xuống, Phải, Lên, Trái) nếu ô đó hợp lệ và chưa đi qua.

2. Kiểm tra (Check)

Nếu đạt đến ô đích (5,5), thông báo thành công. Nếu không tìm thấy đường đi tiếp từ ô hiện tại, đánh dấu ô đó là ngõ cụt.

3. Quay lui (Backtrack)

Rút lui về ô trước đó, gỡ bỏ dấu vết và tiếp tục thử các hướng còn lại chưa xét.

Mã nguồn C tham khảo 

#include <stdio.h>
#define N 6

// Hàm in ma trận đường đi
void printSolution(int sol[N][N]) {
    for (int i = 0; i < N; i++) {
        for (int j = 0; j < N; j++)
            printf("%d ", sol[i][j]);
        printf("\n");
    }
}

// Kiểm tra vị trí (x, y) có hợp lệ trong mê cung hay không
int isSafe(int maze[N][N], int x, int y) {
    return (x >= 0 && x < N && y >= 0 && y < N && maze[x][y] == 0);
}

// Hàm đệ quy giải mê cung bằng quay lui
int solveMazeUtil(int maze[N][N], int x, int y, int sol[N][N]) {
    // Nếu đã đến đích (N-1, N-1), hoàn thành
    if (x == N - 1 && y == N - 1 && maze[x][y] == 0) {
        sol[x][y] = 1;
        return 1;
    }

    // Kiểm tra xem maze[x][y] có phải là ô đi được không
    if (isSafe(maze, x, y) == 1) {
        // Đã đi qua ô này trong đường đi hiện tại
        sol[x][y] = 1;

        // 1. Thử di chuyển xuống dưới
        if (solveMazeUtil(maze, x + 1, y, sol) == 1)
            return 1;

        // 2. Nếu xuống dưới không được, thử di chuyển sang phải
        if (solveMazeUtil(maze, x, y + 1, sol) == 1)
            return 1;

        // 3. Nếu sang phải không được, thử di chuyển lên trên
        if (solveMazeUtil(maze, x - 1, y, sol) == 1)
            return 1;

        // 4. Nếu lên trên không được, thử di chuyển sang trái
        if (solveMazeUtil(maze, x, y - 1, sol) == 1)
            return 1;

        // Nếu không hướng nào đi được -> Quay lui (Backtrack)
        sol[x][y] = 0;
        return 0;
    }

    return 0;
}

int main() {
    int maze[N][N] = {
        {0, 0, 1, 0, 0, 0},
        {1, 0, 1, 0, 1, 0},
        {0, 0, 0, 0, 1, 0},
        {1, 0, 1, 0, 0, 0},
        {0, 1, 0, 1, 1, 0},
        {1, 1, 0, 0, 0, 0}
    };

    int sol[N][N] = { {0} };

    if (solveMazeUtil(maze, 0, 0, sol) == 0) {
        printf("Khong tim thay duong đi!\n");
        return 0;
    }

    printf("Duong di tim thay:\n");
    printSolution(sol);
    return 0;
}
